{% translate "1. Enter the NetBox virtual environment and install the plugin package:" %}

source /opt/netbox/venv/bin/activate

{% translate "2. Add the plugin to the PLUGINS list in configuration.py:" %}

]

{% translate "3. In the virtual environment, run database migrations and collect static files:" %}

python3 /opt/netbox/netbox/manage.py migrate
python3 /opt/netbox/netbox/manage.py collectstatic

{% translate "4. Restart the NetBox services to complete the plugin installation:" %}

sudo systemctl restart netbox netbox-rq
